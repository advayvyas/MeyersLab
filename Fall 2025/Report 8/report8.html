Standard

bart_fit_std = bart(x.train = x.train_std, y.train = y.train_std,
    verbose = FALSE, keeptrees = TRUE)

bart_vi_std = as.data.frame(bart_fit_std$varcount) %>%
    summarise(across(everything(), ~mean(.x, na.rm = TRUE))) %>%
    pivot_longer(cols = everything(), names_to = "variable",
        values_to = "avg")

pred_std = predict(bart_fit_std, newdata = x.test_std)
bart_predictions_std = as.data.frame(t(apply(pred_std, 2, quantile,
    probs = c(0.025, 0.975))))
colnames(bart_predictions_std) = c("ci_low", "ci_high")

bart_predictions_std$estimate = apply(pred_std, 2, mean)
bart_predictions_std$observed = y.test_std

coverage_95_bart_std = mean(bart_predictions_std$observed >=
    bart_predictions_std$ci_low & bart_predictions_std$observed <=
    bart_predictions_std$ci_high, na.rm = TRUE)

ggplot(bart_predictions_std, aes(x = estimate, y = observed)) +
    geom_errorbar(aes(ymin = ci_low, ymax = ci_high), width = 0,
        alpha = 0.35) + geom_point(alpha = 0.7) + geom_abline(slope = 1,
    intercept = 0, linetype = "dashed", color = "red") + labs(x = "Predicted diff_wis_season",
    y = "Observed diff_wis_season", title = "Observed vs Predicted with 95% Prediction Intervals (season)",

New metrics for standard

pred_mean = bart_predictions_std$estimate
obs = bart_predictions_std$observed
n = length(obs)
p = ncol(x.train_std)

R2 = 1 - sum((obs - pred_mean)^2)/sum((obs - mean(obs))^2)
adj_R2 = 1 - (1 - R2) * (n - 1)/(n - p - 1)
RMSE = sqrt(mean((obs - pred_mean)^2))
MAE = mean(abs(obs - pred_mean))
coverage_95 = coverage_95_bart_std

bart_metrics = tibble(R2 = R2, adj_R2 = adj_R2, RMSE = RMSE,
    MAE = MAE, coverage_95 = coverage_95)

Investigating row count mishaps

Looks like I have 833 rows? I have that amount of observations for some reason. Let me check. Alright, socieconomic had duplicate HSA IDs because counties can share HSA IDs. You can run this after, I saved the data to a new file.

nrow(df_complete)
## [1] 173
overall2 = read.csv("../forecasting_metrics_overall.csv")
pop_desc = read.csv("../us_hsa_county_popdesc.csv")
socioecon = read.csv("../Report 7/socioecon.csv")

us_hsa_county_popdesc_bygroup = pop_desc %>%
    select(state, hsa_nci_id, area_km2_hsa, area_km2_state, n_hsa) %>%
    distinct()

overall = overall2 %>%
    left_join(us_hsa_county_popdesc_bygroup, by = c("state",
        "hsa_nci_id"))

df_var = overall %>%
    filter(season == "2023/24", horizon == 3) %>%
    distinct()
socioecon_unique = socioecon %>%
    group_by(hsa_nci_id) %>%
    summarise(across(where(is.numeric), \(x) mean(x,
        na.rm = TRUE)))
df_complete = left_join(df_var, socioecon_unique, by = "hsa_nci_id") %>%
    mutate(density_ratio = density_hsa/density_state, area_ratio = area_km2_hsa/area_km2_state)
write.csv(df_complete, "../complete_metrics.csv", row.names = FALSE)

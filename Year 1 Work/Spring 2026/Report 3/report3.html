Metric functions

Explanations

The discrete_frechet function compares two sequences of points \(P\) and \(Q\) and computes the smallest possible “worst-case” distance needed to match them in order. It first converts the inputs to matrices and builds a full distance matrix dist_mat, where entry \((i,j)\) is the distance between point \(P_i\) and \(Q_j\). It then uses dynamic programming with a table D, where D[i, j] represents the best (smallest possible) maximum distance encountered when matching the prefixes \(P_{1:i}\) and \(Q_{1:j}\). The first row and column are initialized by taking running maxima, since along the edges you can only move forward in one sequence and must keep track of the largest distance seen so far. For each interior cell, the algorithm looks at the three allowed predecessors (up, left, diagonal), chooses the one with the smallest current “worst value,” and then updates D[i, j] as \(\max{D_{\text{prev}}, d(P_i,Q_j)}\). This implements the min–max recursion that defines the discrete Fréchet distance. The Prev array stores which predecessor was chosen so that, after filling the table, the code can backtrack from \((n,m)\) to \((1,1)\) and recover an optimal alignment path. Finally, it extracts the distances along this path and computes their cumulative maxima, showing how the Fréchet value builds up step by step. The returned frechet value is \(D[n,m]\), which is the minimum possible maximum mismatch between the two curves under any valid alignment.

discrete_frechet = function(P, Q, dist_fun = NULL) {
    # P: n x d matrix or data.frame Q: m x d matrix or
    # data.frame dist_fun: function(x, y) returning scalar
    # distance

    if (is.null(dist_fun)) {
        dist_fun = function(x, y) sqrt(sum((x - y)^2))
    }

    P = as.matrix(P)
    Q = as.matrix(Q)

    n = nrow(P)
    m = nrow(Q)

    # distance matrix
    dist_mat = matrix(0, n, m)
    for (i in 1:n) for (j in 1:m) dist_mat[i, j] = dist_fun(P[i,
        ], Q[j, ])

    # DP table
    D = matrix(Inf, n, m)
    Prev = array(NA_integer_, dim = c(n, m, 2))

    # initialization
    D[1, 1] = dist_mat[1, 1]

    for (i in 2:n) {
        D[i, 1] = max(D[i - 1, 1], dist_mat[i, 1])
        Prev[i, 1, ] = c(i - 1, 1)
    }

    for (j in 2:m) {
        D[1, j] = max(D[1, j - 1], dist_mat[1, j])
        Prev[1, j, ] = c(1, j - 1)
    }

    # DP recursion
    for (i in 2:n) {
        for (j in 2:m) {
            candidates = c(D[i - 1, j], D[i, j - 1], D[i - 1,
                j - 1])

            k = which.min(candidates)

            if (k == 1)
                Prev[i, j, ] = c(i - 1, j)
            if (k == 2)
                Prev[i, j, ] = c(i, j - 1)
            if (k == 3)
                Prev[i, j, ] = c(i - 1, j - 1)

            D[i, j] = max(dist_mat[i, j], candidates[k])
        }
    }

    # backtrack optimal path
    i = n
    j = m
    path = matrix(c(i, j), ncol = 2)

    while (!(i == 1 && j == 1)) {
        p = Prev[i, j, ]
        i = p[1]
        j = p[2]
        path = rbind(c(i, j), path)
    }

    # stepwise Frechet values
    step_dist = apply(path, 1, function(idx) dist_mat[idx[1],
        idx[2]])

    frechet_steps = cummax(step_dist)

    list(frechet = D[n, m], path = path, step_distance = step_dist,
        frechet_by_step = frechet_steps)
}

The dtw_distance function also starts by forming a distance matrix between all pairs of points in \(P\) and \(Q\), but instead of focusing on the maximum distance, it focuses on minimizing the total accumulated distance. It creates a dynamic programming table D where D[i, j] represents the minimum cumulative cost of aligning the prefixes \(P_{1:i}\) and \(Q_{1:j}\). The first row and column are initialized by summing distances, since along the boundary the only way to proceed is to keep adding costs. For each interior entry, the function applies the standard DTW recursion \[ D(i,j) = d(P_i,Q_j) + \min{D(i-1,j), D(i,j-1), D(i-1,j-1)}. \] This allows one sequence to “stretch” relative to the other, because horizontal or vertical moves correspond to repeating points in one series. After filling D, the algorithm backtracks from \((n,m)\) to \((1,1)\) by always moving to the neighboring cell with the smallest cumulative cost, thereby reconstructing the optimal warping path. It then extracts the distances along this path and computes their cumulative sum, showing how the total cost grows. The final dtw_distance value is \(D[n,m]\), which represents the smallest possible total mismatch between the two sequences under time warping.

dtw_distance = function(P, Q, dist_fun = NULL) {
    # P, Q: numeric vectors or matrices (rows = time
    # points)
    if (is.null(dist_fun))
        dist_fun = function(x, y) sqrt(sum((x - y)^2))

    P = as.matrix(P)
    Q = as.matrix(Q)

    n = nrow(P)
    m = nrow(Q)

    # distance matrix
    dist_mat = matrix(0, n, m)
    for (i in 1:n) for (j in 1:m) dist_mat[i, j] = dist_fun(P[i,
        ], Q[j, ])

    # cumulative cost matrix
    D = matrix(Inf, n, m)
    D[1, 1] = dist_mat[1, 1]

    # initialize first row/column
    for (i in 2:n) D[i, 1] = dist_mat[i, 1] + D[i - 1, 1]
    for (j in 2:m) D[1, j] = dist_mat[1, j] + D[1, j - 1]

    # fill rest of matrix
    for (i in 2:n) {
        for (j in 2:m) {
            D[i, j] = dist_mat[i, j] + min(D[i - 1, j], D[i,
                j - 1], D[i - 1, j - 1])
        }
    }

    # backtrack optimal path
    i = n
    j = m
    path = matrix(c(i, j), ncol = 2)
    while (!(i == 1 && j == 1)) {
        if (i == 1) {
            j = j - 1
        } else if (j == 1) {
            i = i - 1
        } else {
            # choose minimum predecessor
            k = which.min(c(D[i - 1, j - 1], D[i - 1, j], D[i,
                j - 1]))
            if (k == 1) {
                i = i - 1
                j = j - 1
            } else if (k == 2) {
                i = i - 1
            } else {
                j = j - 1
            }
        }
        path = rbind(c(i, j), path)
    }

    # stepwise cumulative cost along path
    step_cost = dist_mat[path]
    cum_cost = cumsum(step_cost)

    list(dtw_distance = D[n, m], path = path, step_cost = step_cost,
        cum_cost = cum_cost)
}
graph_hsa = function(hsa_id = 688) {
    hsa = hsa_state[hsa_state$hsa_nci_id == hsa_id, ]

    P = as.matrix(hsa$inc_hsa)
    Q = as.matrix(hsa$inc_state)

    res_dtw = dtw_distance(P, Q)

    res = discrete_frechet(P, Q)

    normal = ggplot(hsa) + geom_line(linewidth = 1, (aes(x = seq_len(148),
        y = inc_hsa, color = "HSA"))) + geom_line(linewidth = 1,
        (aes(x = seq_len(148), y = inc_state, color = "State"))) +
        labs(x = "index (weeks, 2022 Week 39 to 2025 Week 30)",
            y = "infected_value (?)", title = paste0("Local vs. state inc values for HSA ID ",
                hsa_id, " from ", hsa$state)) + theme_minimal() +
        scale_color_manual(name = "Legend", values = c(HSA = "blue",
            State = "red"))

    frechet = ggplot() + geom_line(aes(x = seq_len(length(res$frechet_by_step)),
        y = res$frechet_by_step), linewidth = 1) + theme_minimal() +
        labs(x = "Steps", y = "Frechet distance", title = "Frechet distance by step")

    dtw = ggplot() + geom_line(aes(x = seq_len(length(res_dtw$cum_cost)),
        y = res_dtw$cum_cost), linewidth = 1) + theme_minimal() +
        labs(x = "Steps", y = "Cumulative DTW cost", title = "DTW cumulative cost along optimal path by step")

    # dtw adjusted
    P_warped = rep(NA, length(Q))
    i = res_dtw$path[, 1]
    j = res_dtw$path[, 2]
    P_warped[j] = P[i]
    # P_warped = na.approx(P_warped, rule = 2)

    df_plot = data.frame(index = seq_len(length(Q)), HSA_aligned = P_warped,
        State = Q)

    warped = ggplot(df_plot) + geom_line(aes(x = index, y = HSA_aligned,
        color = "HSA (warped)"), linewidth = 1) + geom_line(aes(x = index,
        y = State, color = "State"), linewidth = 1) + labs(x = "index (weeks, 2022 Week 39 to 2025 Week 30)",
        y = "infected_value (?)", title = paste0("DTW-aligned Local vs. state inc values for HSA ID ",
            hsa_id, " from ", hsa$state)) + theme_minimal() +
        scale_color_manual(name = "Legend", values = c(`HSA (warped)` = "blue",
            State = "red"))

    (normal/warped)/(frechet + dtw)
}
dtw_stats = function(hsa_id = 688) {
    hsa = hsa_state[hsa_state$hsa_nci_id == hsa_id, ]

    P = as.matrix(hsa$inc_hsa)
    Q = as.matrix(hsa$inc_state)

    res_dtw = dtw_distance(P, Q)

    res = discrete_frechet(P, Q)

    i = res_dtw$path[, 1]
    j = res_dtw$path[, 2]

    delta_t = j - i
    delta_x = P[i, 1] - Q[j, 1]

    stats_list = list(mean_abs_lag = mean(abs(delta_t)), signed_lag = mean(delta_t),
        lag_sd = sd(delta_t), dtw_mae = mean(abs(delta_x)), dtw_rmse = sqrt(mean(delta_x^2)),
        rmse = sqrt(mean((P[, 1] - Q[, 1])^2)), amp_ratio = median(P[,
            1]/Q[, 1], na.rm = TRUE), dtw_amp_ratio = median(P[i,
            1]/Q[j, 1], na.rm = TRUE), lag_amp_corr = cor(abs(delta_t),
            abs(delta_x)), frechet = res$frechet)

    stats = data.frame(Metric = names(stats_list), Value = unname(unlist(stats_list)))

    return(stats)
}

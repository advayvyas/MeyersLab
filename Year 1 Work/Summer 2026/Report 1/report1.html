truth_lag_data = diff_peak_data %>%
    select(ID, peak_window, hsa_peak_week, state_peak_week) %>%
    mutate(truth_lag = hsa_peak_week - state_peak_week)

timing_summary = prob_results_window %>%
    left_join(truth_lag_data, by = c("ID", "peak_window")) %>%
    group_by(peak_window) %>%
    summarise(n = n(), mean_lag_mean = mean(mean_lag, na.rm = TRUE),
        mean_lag_sd = sd(mean_lag, na.rm = TRUE), mean_lag_se = mean_lag_sd/sqrt(n),
        sd_lag_mean = mean(sd_lag, na.rm = TRUE), sd_lag_sd = sd(sd_lag,
            na.rm = TRUE), sd_lag_se = sd_lag_sd/sqrt(n), truth_lag_mean = mean(truth_lag,
            na.rm = TRUE), .groups = "drop") %>%
    mutate(peak_window = recode(peak_window, `1` = "2022/23",
        `2` = "2023/24", `3` = "2024/25")) %>%
    mutate(mean_lag_ci_lower = mean_lag_mean - 1.96 * mean_lag_se,
        mean_lag_ci_upper = mean_lag_mean + 1.96 * mean_lag_se,
        sd_lag_ci_lower = sd_lag_mean - 1.96 * sd_lag_se, sd_lag_ci_upper = sd_lag_mean +
            1.96 * sd_lag_se)

Mean Lag (weeks)

This is the average signed timing offset. Positive = HSA leads state; negative = HSA lags.

timing_for_plot = timing_summary %>%

Stratified by Population Ratio

pop_data = raw_data %>%
    select(hsa_nci_id, season, pop_ratio) %>%
    distinct()

season_map = tibble(peak_window = c(1, 2, 3), season = c("2022/23",
    "2023/24", "2024/25"))

prob_results_pop = prob_results_window %>%
    left_join(season_map, by = "peak_window") %>%
    left_join(pop_data, by = c(ID = "hsa_nci_id", season = "season")) %>%
    filter(!is.na(pop_ratio)) %>%
    mutate(pop_ratio_q = ntile(pop_ratio, 4))

timing_by_pop = prob_results_pop %>%
    group_by(peak_window, pop_ratio_q) %>%
    summarise(n = n(), mean_lag_mean = mean(mean_lag, na.rm = TRUE),
        mean_lag_se = sd(mean_lag, na.rm = TRUE)/sqrt(n), mean_lag_ci_lower = mean_lag_mean -
            1.96 * mean_lag_se, mean_lag_ci_upper = mean_lag_mean +
            1.96 * mean_lag_se, .groups = "drop") %>%
    mutate(peak_window = recode(peak_window, `1` = "2022/23",
        `2` = "2023/24", `3` = "2024/25"), pop_ratio_q = recode(pop_ratio_q,
        `1` = "Q1 (smallest)", `2` = "Q2", `3` = "Q3", `4` = "Q4 (largest)"))

ggplot(timing_by_pop, aes(x = peak_window, y = mean_lag_mean,
    color = pop_ratio_q)) + geom_point(size = 3, position = position_dodge(width = 0.3)) +
    geom_errorbar(aes(ymin = mean_lag_ci_lower, ymax = mean_lag_ci_upper),
        width = 0.15, position = position_dodge(width = 0.3),
        lwd = 0.8) + geom_hline(yintercept = 0, linetype = "dashed",
    color = "gray50") + labs(title = "Mean Lag by Season and Population Ratio Quartile",
    x = "Season", y = "Mean lag (weeks)", color = "Population ratio") +
    theme_bw() + theme(legend.position = "right")

Conclusion

Timing metrics look interesting, I think soft-DTW lag is still backwards or ground truth is backwards. Magnitude metrics look interesting, the ratios look nice but I’m having trouble seeing if the trends actually match. I like the quantile plot a lot and it looks pretty useful, describes how HSAs with more similar populations have smaller lags.

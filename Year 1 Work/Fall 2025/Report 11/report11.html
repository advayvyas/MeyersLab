XGBoost

dtrain = xgb.DMatrix(data = as.matrix(x.train), label = y.train)
dtest  = xgb.DMatrix(data = as.matrix(x.test), label = y.test)

params = list(
  objective = "reg:squarederror",
  eval_metric = "rmse",
  eta = 0.05,              # lower learning rate for smoother fits
  max_depth = 2,           # shallow trees for low variance
  subsample = 0.6,         # randomly drop rows
  colsample_bytree = 0.6,  # randomly drop columns
  min_child_weight = 5,    # require decent sample size per leaf
  lambda = 5,              # L2 regularization
  alpha = 2                # L1 regularization
)

cv = xgb.cv(
  params = params,
  data = dtrain,
  nrounds = 500,
  nfold = 10,
  early_stopping_rounds = 20,
  verbose = 0
)

best_nrounds = cv$best_iteration

xgb_fit = xgb.train(
  params = params,
  data = dtrain,
  nrounds = best_nrounds,
  verbose = 0
)


pred_mean = predict(xgb_fit, as.matrix(x.test))

n = length(y.test)
p = ncol(x.train)
R2 = 1 - sum((y.test - pred_mean)^2) / sum((y.test - mean(y.test))^2)

RMSE = sqrt(mean((y.test - pred_mean)^2))

cv_mean_R2 = 1 - (cv$evaluation_log$test_rmse_mean[cv$best_iteration]^2 /
                  var(y.train))

metrics = data.frame (
  Metric = c("$R^2$", "Cross-validated $R^2$", "RMSE"),
  Value = round(c(R2, cv_mean_R2, RMSE),4)
)

Horizon switches

overall2 = read.csv("../forecasting_metrics_overall.csv")
pop_desc = read.csv("../us_hsa_county_popdesc.csv")
socioecon = read.csv("../Report 7/socioecon.csv")

us_hsa_county_popdesc_bygroup = pop_desc %>%
    select(state, hsa_nci_id, area_km2_hsa, area_km2_state, n_hsa) %>%
    distinct()

overall = overall2 %>%
    left_join(us_hsa_county_popdesc_bygroup, by = c("state",
        "hsa_nci_id"))

df_var_horizon = overall %>%
    filter(season == "2023/24") %>%
    distinct()
socioecon_unique = socioecon %>%
    group_by(hsa_nci_id) %>%
    summarise(across(where(is.numeric), \(x) mean(x,
        na.rm = TRUE)))
df_horizon = left_join(df_var_horizon, socioecon_unique, by = "hsa_nci_id") %>%
    mutate(density_ratio = density_hsa/density_state, area_ratio = area_km2_hsa/area_km2_state)
write.csv(df_horizon, "../all_horizon_complete_metrics.csv",
    row.names = FALSE)

df_horizon = drop_na(df_horizon)

Horizon 1

df_horizon_1 = df_horizon[df_horizon$horizon == 1, ]

n = nrow(df_horizon_1)
train_idx = sample(seq_len(n), size = floor(0.8 * n))  # 80% train
test_idx = setdiff(seq_len(n), train_idx)

x.train = df_horizon_1[train_idx, vars]
y.train = df_horizon_1$diff_wis_season[train_idx]

x.test = df_horizon_1[test_idx, vars]
y.test = df_horizon_1$diff_wis_season[test_idx]

bart_fit = bart(x.train = x.train, y.train = y.train, verbose = FALSE,
    keeptrees = TRUE)

pred = predict(bart_fit, newdata = x.test)
bart_predictions = as.data.frame(t(apply(pred, 2, quantile, probs = c(0.025,
    0.975))))
colnames(bart_predictions) = c("ci_low", "ci_high")

bart_predictions$estimate = apply(pred, 2, mean)
bart_predictions$observed = y.test

coverage_95_bart = mean(bart_predictions$observed >= bart_predictions$ci_low &
    bart_predictions$observed <= bart_predictions$ci_high, na.rm = TRUE)

folds = createFolds(y.train, k = 5)

cv_R2 = c()

for (i in seq_along(folds)) {
    test_idx = folds[[i]]
    train_idx = setdiff(seq_along(y.train), test_idx)

    bart_fit_cv = bart(x.train = x.train[train_idx, ], y.train = y.train[train_idx],
        verbose = FALSE, keeptrees = TRUE)

    pred_cv = colMeans(predict(bart_fit_cv, newdata = x.train[test_idx,
        ]))
    y_true = y.train[test_idx]

    R2_fold = 1 - sum((y_true - pred_cv)^2)/sum((y_true - mean(y_true))^2)
    cv_R2[i] = R2_fold
}

cv_R2_bart = mean(cv_R2)

R2_bart = 1 - sum((bart_predictions$observed - bart_predictions$estimate)^2)/sum((bart_predictions$observed -
    mean(bart_predictions$observed))^2)

rmse_bart = sqrt(mean((y.test - colMeans(pred))^2))

horizon_1 = c(R2_bart, cv_R2_bart, rmse_bart, coverage_95_bart)
ggplot(bart_predictions, aes(x = estimate, y = observed)) + geom_errorbar(aes(ymin = ci_low,
    ymax = ci_high), width = 0, alpha = 0.35) + geom_point(alpha = 0.7) +
    geom_abline(slope = 1, intercept = 0, linetype = "dashed",
        color = "red") + labs(x = "Predicted diff_wis_season",
    y = "Observed diff_wis_season", title = "Observed vs Predicted with 95% Prediction Intervals (season, socioeconomic)",
    subtitle = sprintf("95%% coverage = %.1f%% (n=%d)", 100 *
        coverage_95_bart, nrow(bart_predictions))) + theme_minimal() +
    scale_x_continuous(breaks = seq(-1, 2, by = 0.25)) + coord_cartesian(xlim = c(-0.25,
    1))
